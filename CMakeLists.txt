cmake_minimum_required(VERSION 3.21)
project(DeepMirrorCapture VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets OpenGLWidgets Concurrent)
qt_standard_project_setup()
qt_add_executable(deepmirror src/main.cpp src/capture.cpp src/capture.h src/processing.cpp src/processing.h)
set_target_properties(deepmirror PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER com.deepmirror.capture WIN32_EXECUTABLE TRUE)
target_link_libraries(deepmirror PRIVATE Qt6::Widgets Qt6::OpenGLWidgets Qt6::Concurrent)
enable_testing()
add_test(NAME capture_roundtrip COMMAND deepmirror --self-test ${CMAKE_CURRENT_BINARY_DIR}/test-output)
set_tests_properties(capture_roundtrip PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

option(DEEPMIRROR_ENABLE_WINDOWS_SDK "Build Windows camera backend" ${WIN32})
if(WIN32 AND DEEPMIRROR_ENABLE_WINDOWS_SDK)
 if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Camera SDK requires an MSVC x64 Qt kit")
 endif()
 include(cmake/InsightSDK.cmake)
else()
 target_sources(deepmirror PRIVATE src/windows_stub.cpp)
endif()
if(MSVC)
 target_compile_options(deepmirror PRIVATE /utf-8 /EHsc)
 target_compile_definitions(deepmirror PRIVATE NOMINMAX)
endif()
qt_add_executable(processing_test tests/processing_test.cpp src/processing.cpp src/capture.cpp)
target_include_directories(processing_test PRIVATE src)
target_link_libraries(processing_test PRIVATE Qt6::Gui)
add_test(NAME rgbd_processing COMMAND processing_test)
set_tests_properties(rgbd_processing PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
